추가 셀 1-1. Alibaba 사용자 코드 로컬 통일 (오류 방지, 필수)
-------------------------------------------------------------

import os
import json
import shutil
from huggingface_hub import hf_hub_download

REVISION = '40ced75c3017eb27626c9d4ea981bde21a2662f4'

def localize_alibaba_checkpoint(checkpoint):
    # 설정 코드와 모델 코드를 같은 체크포인트 폴더에서 읽도록 통일합니다.
    for filename in ['configuration.py', 'modeling.py']:
        source = hf_hub_download(
            repo_id='Alibaba-NLP/new-impl',
            filename=filename,
            revision=REVISION,
        )
        shutil.copy2(source, os.path.join(checkpoint, filename))

    config_path = os.path.join(checkpoint, 'config.json')
    with open(config_path, encoding='utf-8') as f:
        config = json.load(f)

    config['auto_map'] = {
        'AutoConfig': 'configuration.NewConfig',
        'AutoModel': 'modeling.NewModel',
        'AutoModelForMaskedLM': 'modeling.NewForMaskedLM',
        'AutoModelForMultipleChoice': 'modeling.NewForMultipleChoice',
        'AutoModelForQuestionAnswering': 'modeling.NewForQuestionAnswering',
        'AutoModelForSequenceClassification': 'modeling.NewForSequenceClassification',
        'AutoModelForTokenClassification': 'modeling.NewForTokenClassification',
    }
    with open(config_path, 'w', encoding='utf-8') as f:
        json.dump(config, f, ensure_ascii=False, indent=2)

    # 앞서 서로 다르게 로드된 사용자 코드 캐시를 제거합니다.
    cache_dir = os.path.join(
        '/root/.cache/huggingface/modules/transformers_modules',
        os.path.basename(checkpoint),
    )
    shutil.rmtree(cache_dir, ignore_errors=True)
    print('사용자 코드 통일 완료:', checkpoint)

localize_alibaba_checkpoint('/content/drive/MyDrive/alibaba_v2_full')
